[CmdletBinding()]
param()
$ErrorActionPreference = 'Stop'
$repo = [IO.Path]::GetFullPath((Join-Path $PSScriptRoot '../../../..'))
$commit = '2d79ba55fae1b1e71c69f93a668080c9029e6a31'
$archive = Join-Path $repo "artifacts/portability/native-host/avalonia-$commit.tar.gz"
$archiveHash = '9E9EE6EE270BFDC02FBC34296DA51EE6242325156D2B67CBED92D2CCBC2E2029'
if ((Get-FileHash -LiteralPath $archive -Algorithm SHA256).Hash -ne $archiveHash) { throw 'Pinned archive mismatch.' }
$unicode = @('BiDi.trie.cs','BiDiAlgorithm.cs','BiDiClass.cs','BiDiData.cs','BiDiPairedBracketType.cs','Codepoint.cs','CodepointEnumerator.cs','EastAsianWidth.trie.cs','EastAsianWidthClass.cs','GeneralCategory.cs','Grapheme.cs','GraphemeBreak.trie.cs','GraphemeBreakClass.cs','GraphemeEnumerator.cs','IndicConjunctBreakClass.cs','LineBreak.cs','LineBreakClass.cs','LineBreakEnumerator.cs','Script.cs','ScriptExtensions.data.cs','UnicodeData.cs','UnicodeData.Lookups.cs','UnicodeData.trie.cs','UnicodeDataSource.cs','UnicodeTrie.cs','UnicodeTrieBuilder.Constants.cs','Utf16Utils.cs','WordBreakClass.cs')
$paths = @($unicode | ForEach-Object { "src/Avalonia.Base/Media/TextFormatting/Unicode/$_" })
$paths += @('ArrayBuilder.cs','ArraySlice.cs','MappedArraySlice.cs','ImmutableReadOnlyListStructEnumerator.cs','BidiDictionary.cs') | ForEach-Object { "src/Avalonia.Base/Utilities/$_" }
$paths += 'src/Avalonia.Base/Media/TextFormatting/FormattingBufferHelper.cs'
$paths += @('licence.md','NOTICE.md')
$paths += @('tests/Avalonia.Skia.UnitTests/Fonts/DejaVuSans.ttf','src/Avalonia.Fonts.Inter/Assets/Inter-Regular.ttf')
Add-Type -AssemblyName System.Formats.Tar
$file = [IO.File]::OpenRead($archive)
$gzip = [IO.Compression.GZipStream]::new($file, [IO.Compression.CompressionMode]::Decompress)
$reader = [System.Formats.Tar.TarReader]::new($gzip)
$manifest = @()
try {
    while ($entry = $reader.GetNextEntry()) {
        $relative = $entry.Name.Substring($entry.Name.IndexOf('/') + 1)
        if ($relative -notin $paths) { continue }
        $destination = Join-Path $PSScriptRoot "Original/$relative"
        [IO.Directory]::CreateDirectory([IO.Path]::GetDirectoryName($destination)) | Out-Null
        $output = [IO.File]::Create($destination)
        try { $entry.DataStream.CopyTo($output) } finally { $output.Dispose() }
        $content = if ($relative.EndsWith('.cs')) { [IO.File]::ReadAllText($destination) } else { '' }
        $publicCount = [regex]::Matches($content, '(?m)^    public (?=(?:(?:readonly|ref|partial|sealed|static) )*(?:class|struct|record|enum))').Count
        $namespaceCount = [regex]::Matches($content, 'Avalonia\.(?:Media\.TextFormatting(?:\.Unicode)?|Utilities)').Count
        $license = if ($content.Contains('Licensed under the Apache License')) { 'Apache-2.0' } else { 'MIT' }
        if ($relative.EndsWith('/DejaVuSans.ttf')) { $license = 'Bitstream-Vera AND Arev' }
        if ($relative.EndsWith('/Inter-Regular.ttf')) { $license = 'OFL-1.1' }
        if ($relative.Contains('.trie.') -or $relative.Contains('ScriptExtensions.data')) { $license += ' AND Unicode-3.0' }
        $builderCount = if ($relative.EndsWith('/ArrayBuilder.cs')) { 0 } else { [regex]::Matches($content, '\bArrayBuilder<').Count }
        $manifest += [ordered]@{ path=$relative; sha256=(Get-FileHash -LiteralPath $destination -Algorithm SHA256).Hash; license=$license; namespaces=$namespaceCount; publicTypes=$publicCount; qualifiedBuilders=$builderCount }
    }
} finally { $reader.Dispose(); $gzip.Dispose(); $file.Dispose() }
if ($manifest.Count -ne $paths.Count) { throw 'The pinned archive is missing a requested file.' }
[IO.File]::WriteAllText((Join-Path $PSScriptRoot 'manifest.json'), ([ordered]@{ commit=$commit; archiveSha256=$archiveHash; unicodeVersion='17.0.0'; files=@($manifest | Sort-Object path) } | ConvertTo-Json -Depth 5) + "`n")
$xml = [Xml.XmlDocument]::new()
$project = $xml.CreateElement('Project')
$xml.AppendChild($project) | Out-Null
$group = $xml.CreateElement('ItemGroup')
$project.AppendChild($group) | Out-Null
foreach ($record in ($manifest | Sort-Object path)) {
    $item = $xml.CreateElement('TextUnicodeSource')
    $item.SetAttribute('Include', '$(MSBuildThisFileDirectory)Original/' + $record.path)
    foreach ($key in @('sha256','namespaces','publicTypes','qualifiedBuilders')) { $item.SetAttribute($key, [string]$record[$key]) }
    $group.AppendChild($item) | Out-Null
}
$xml.Save((Join-Path $PSScriptRoot 'Sources.props'))
foreach ($fontRecord in ($manifest | Where-Object { $_.path.EndsWith('.ttf') })) {
    $fontBytes = [IO.File]::ReadAllBytes((Join-Path $PSScriptRoot ('Original/' + $fontRecord.path)))
    function Read-U16($p) { return [int]$fontBytes[$p] * 256 + [int]$fontBytes[$p + 1] }
    function Read-U32($p) { return [long]$fontBytes[$p] * 16777216 + [long]$fontBytes[$p + 1] * 65536 + [long]$fontBytes[$p + 2] * 256 + [long]$fontBytes[$p + 3] }
    $nameOffset = -1
    for ($i = 0; $i -lt (Read-U16 4); $i++) {
        $p = 12 + 16 * $i
        if ([Text.Encoding]::ASCII.GetString($fontBytes, $p, 4) -eq 'name') { $nameOffset = Read-U32 ($p + 8) }
    }
    if ($nameOffset -lt 0) { throw 'Font name table is absent.' }
    $stringOffset = $nameOffset + (Read-U16 ($nameOffset + 4))
    $notices = @()
    for ($i = 0; $i -lt (Read-U16 ($nameOffset + 2)); $i++) {
        $p = $nameOffset + 6 + 12 * $i
        $id = Read-U16 ($p + 6)
        if (($id -in @(0, 13, 14)) -and (Read-U16 $p) -eq 3 -and (Read-U16 ($p + 4)) -eq 1033) {
            $notices += [Text.Encoding]::BigEndianUnicode.GetString($fontBytes, $stringOffset + (Read-U16 ($p + 10)), (Read-U16 ($p + 8)))
        }
    }
    if ($notices.Count -lt 2) { throw 'Font copyright/license notices are incomplete.' }
    $name = if ($fontRecord.path.EndsWith('/DejaVuSans.ttf')) { 'DejaVu' } else { 'Inter-Copyright' }
    $licenseDirectory = Join-Path $PSScriptRoot 'Licenses'
    [IO.Directory]::CreateDirectory($licenseDirectory) | Out-Null
    [IO.File]::WriteAllText((Join-Path $licenseDirectory ($name + '.txt')), ($notices -join "`n`n") + "`n")
}
